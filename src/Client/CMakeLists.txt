kobalt_add_lib(Client)

target_link_libraries(Client PRIVATE Base)
target_link_libraries(Client PRIVATE Engine)
target_link_libraries(Client PRIVATE GpuInterface)
target_link_libraries(Client PUBLIC Renderer) #PUBLIC: Client.h holds a Renderer by value
target_link_libraries(Client PUBLIC SDL3::SDL3)
target_link_libraries(Client PRIVATE ${Vulkan_LIBRARIES})

# Specify the include directories so that other projects can use this module's headers
target_include_directories(Client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_compile_definitions(Client PRIVATE CLIENT_DLL_EXPORTS)

# Generated header that includes every client entity header, so the reflection scan can see them.
# Written to the build dir: this subdirectory's binary dir is the source dir.
file(GLOB_RECURSE CLIENT_ENTITY_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Entities/*.h")
list(FILTER CLIENT_ENTITY_HEADERS EXCLUDE REGEX "EntityRegistration\\.h$")

set(CLIENT_ENTITY_INCLUDES "")
foreach(header IN LISTS CLIENT_ENTITY_HEADERS)
	file(RELATIVE_PATH relativeHeader "${ENGINE_SRC}" "${header}")
	string(APPEND CLIENT_ENTITY_INCLUDES "#include \"${relativeHeader}\"\n")
endforeach()

set(CLIENT_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated/Client")
file(CONFIGURE OUTPUT "${CLIENT_GENERATED_DIR}/Generated/ClientEntityHeaders.h"
	CONTENT "#pragma once\n// Generated by CMake from src/Client/Entities, do not edit\n${CLIENT_ENTITY_INCLUDES}"
	@ONLY)

target_include_directories(Client PRIVATE "${CLIENT_GENERATED_DIR}")

set_target_properties(Client PROPERTIES FOLDER Engine)
